class TestRecordAndDrain:
    """Low-level tests on cuda_ops.record_completion_on_stream / drain."""

    def test_drain_empty(self):
        completions = cuda_ops.drain_recorded_completions()
        assert completions == []

    def test_single_completion_payload_is_bytes(self, stream):
        # drain must hand back py::bytes, not utf-8-decoded str — msgpack
        # output is arbitrary bytes including invalid utf-8.
        encoded = msgspec.msgpack.encode([b"key-1", b"key-2"])
        cuda_ops.record_completion_on_stream(stream.ptr, "finish_write", encoded)
        stream.synchronize()

        completions = cuda_ops.drain_recorded_completions()
        assert len(completions) == 1
        kind, payload = completions[0]
        assert kind == "finish_write"
        assert isinstance(payload, bytes)
        assert msgspec.msgpack.decode(payload, type=list[bytes]) == [
            b"key-1",
            b"key-2",
        ]

    def test_many_completions_in_order(self, stream):
        for i in range(50):
            cuda_ops.record_completion_on_stream(
                stream.ptr, "finish_write", msgspec.msgpack.encode(i)
            )
        stream.synchronize()

        completions = cuda_ops.drain_recorded_completions()
        assert len(completions) == 50
        for idx, (kind, payload) in enumerate(completions):
            assert kind == "finish_write"
            assert msgspec.msgpack.decode(payload, type=int) == idx


class TestDispatcher:
    """Integration tests for DeviceHostFuncDispatcher (drain + dispatch)."""

    def test_dispatch_to_registered_handler(self, dispatcher, stream):
        seen: list[list[bytes]] = []
        dispatcher.register("finish_write", seen.append, payload_type=list[bytes])

        submit_callback_to_stream(stream, "finish_write", [b"k0", b"k1", b"k2"])
        stream.synchronize()

        deadline = time.monotonic() + 5.0
        while time.monotonic() < deadline and not seen:
            time.sleep(0.01)
        assert seen == [[b"k0", b"k1", b"k2"]]

    def test_unknown_kind_drops_payload(self, dispatcher, stream):
        submit_callback_to_stream(stream, "finish_unknown", [b"x"])
        stream.synchronize()
        time.sleep(0.1)
        assert dispatcher.dispatched_count() == 0

    def test_handler_exception_does_not_kill_thread(self, dispatcher, stream):
        calls: list[list[bytes]] = []

        def handler(payload):
            calls.append(payload)
            if len(calls) == 1:
                raise RuntimeError("boom")

        dispatcher.register("finish_write", handler, payload_type=list[bytes])
        submit_callback_to_stream(stream, "finish_write", [b"a"])
        submit_callback_to_stream(stream, "finish_write", [b"b"])
        stream.synchronize()

        deadline = time.monotonic() + 5.0
        while time.monotonic() < deadline and len(calls) < 2:
            time.sleep(0.01)
        assert calls == [[b"a"], [b"b"]]
        assert dispatcher.handler_exception_counts().get("finish_write", 0) == 1

    def test_tuple_payload_for_multiple_arguments(self, dispatcher, stream):
        # Multi-arg handlers: wrap args in a tuple, register tuple decode type.
        seen: list[tuple[int, str]] = []
        dispatcher.register("finish_multi", seen.append, payload_type=tuple[int, str])

        submit_callback_to_stream(stream, "finish_multi", (42, "hello"))
        stream.synchronize()

        deadline = time.monotonic() + 5.0
        while time.monotonic() < deadline and not seen:
            time.sleep(0.01)
        assert seen == [(42, "hello")]


class TestDeadlockRegression:
    """Regression for the GIL deadlock: the legacy
    ``stream.launch_host_func(python_fn, ...)`` path stalls; the C++
    callback runs without the GIL and must finish within the timeout."""

    def test_many_concurrent_records_no_deadlock(self, dispatcher, stream):
        received: list[int] = []
        ready = threading.Event()

        def handler(payload):
            received.append(payload)
            if len(received) >= 200:
                ready.set()

        dispatcher.register("finish_write", handler, payload_type=int)

        for i in range(200):
            submit_callback_to_stream(stream, "finish_write", i)
        stream.synchronize()

        assert ready.wait(timeout=10.0), (
            f"deadlock or drop: only {len(received)} of 200 dispatched"
        )
        assert len(received) == 200


@pytest.mark.parametrize("unfull", [False, True])
@pytest.mark.parametrize("release", ["ttl", "free", "peer"])
def test_native_transfer_keeps_read_buffer_until_completion(stream, release, unfull):
    """An actual pending H2D copy must survive read expiry and ordinary eviction."""
    from lmcache.v1.distributed.api import MemoryLayoutDesc, ipc_key_to_object_keys
    from lmcache.v1.distributed.config import (
        EvictionConfig,
        L1ManagerConfig,
        L1MemoryManagerConfig,
        StorageManagerConfig,
    )
    from lmcache.v1.distributed.storage_manager import L1ReadTransferCompletion, l1_owners_of
    from lmcache.v1.common.custom_types import IPCCacheServerKey
    from lmcache.v1.multiprocess.engine_context import MPCacheServerContext
    from lmcache.v1.multiprocess.modules.lookup import LookupModule

    delay = cupy.RawKernel(
        r"""
    extern "C" __global__ void bounded_delay(unsigned long long ns) {
        unsigned long long start, now;
        asm volatile("mov.u64 %0, %%globaltimer;" : "=l"(start));
        do {
            asm volatile("mov.u64 %0, %%globaltimer;" : "=l"(now));
        } while (now - start < ns);
    }
    """,
        "bounded_delay",
    )
    delay.compile()
    context = MPCacheServerContext(
        StorageManagerConfig(
            l1_manager_config=L1ManagerConfig(
                memory_config=L1MemoryManagerConfig(
                    size_in_bytes=2 << 20, use_lazy=False, shm_name=""
                ),
                read_ttl_seconds=1 if release == "ttl" else 30,
            ),
            eviction_config=EvictionConfig(eviction_policy="noop"),
        ),
        chunk_size=4,
    )
    sm = context.storage_manager
    lookup = LookupModule(context)
    dispatch = DeviceHostFuncDispatcher()
    dispatch.register(
        "finish_read_transfer",
        sm.finish_read_transfer,
        payload_type=L1ReadTransferCompletion,
    )
    layout = MemoryLayoutDesc(shapes=[torch.Size([4])], dtypes=[torch.uint8])
    context.layout_desc_registry.register("native-pin", 1, layout)
    key = IPCCacheServerKey.from_token_ids(
        model_name="native-pin",
        world_size=1,
        worker_id=None,
        token_ids=(1, 2, 3, 4, 5, 6, 7) if unfull else (1, 2, 3, 4),
        start=0,
        end=7 if unfull else 4,
        request_id="native-pin-A",
    )
    hashes = context.token_hasher.compute_packed_chunk_hashes(key.token_bytes)
    replication = context.layout_desc_registry.find_shard_replication_desc("native-pin", 1)
    keys = ipc_key_to_object_keys(key, hashes, [0], replication)[0]
    if unfull:
        full = sm.reserve_write(keys, layout)
        sm.finish_write(keys, l1_owners_of(full))
        session = context.session_manager.get_or_create(key.request_id)
        session.set_tokens(key.token_bytes)
        tail_hash = context.token_hasher.hash_to_bytes(session.get_unfull_chunk_hash(key.end))
        keys = ipc_key_to_object_keys(key, [tail_hash], [0], replication)[0]
    try:
        reserved = sm.reserve_write(keys, layout)
        tensor = reserved[keys[0]].tensor
        assert tensor is not None
        tensor.fill_(53)
        sm.finish_write(keys, l1_owners_of(reserved))
        device = cupy.empty(4, dtype=cupy.uint8)
        finished = cupy.cuda.Event()
        lookup.lookup(key, 1)
        assert lookup.wait_prefetch_status(key.request_id, timeout=5) == key.end
        session = context.session_manager.get(key.request_id)
        if release == "peer":
            peer = replace(key, request_id="native-pin-B")
            lookup.lookup(peer, 1)
            assert lookup.wait_prefetch_status(peer.request_id, timeout=5) == peer.end
        claims = session.claim_worker_read(replace(key, worker_id=0), 0, keys)
        assert claims is not None and len(claims) == 1
        completion = sm.prepare_read_generation_completion(claims)
        pinned = sm.pin_read_buffers(completion, "native-transfer-A")
        assert list(pinned) == keys
        # One thread delays only this stream, bounded to 2.5 seconds.
        delay((1,), (1,), (cupy.uint64(2_500_000_000),), stream=stream)
        cupy.cuda.runtime.memcpyAsync(
            device.data.ptr,
            tensor.data_ptr(),
            4,
            cupy.cuda.runtime.memcpyHostToDevice,
            stream.ptr,
        )
        submit_callback_to_stream(
            stream, "finish_read_transfer", ("native-transfer-A", completion)
        )
        finished.record(stream)
        if release == "ttl":
            time.sleep(1.15)
        else:
            lookup.free_lookup_locks(key, 1)
        assert not finished.done, "copy must still be pending during eviction"
        assert sm.delete_l1_keys(keys) == (0, 1)
        sm.clear()
        assert pinned[keys[0]].is_valid()
        stream.synchronize()
        assert device.get().tolist() == [53] * 4
        dispatch.stop()
        assert dispatch.handler_exception_counts() == {}
        assert dispatch.dispatched_count() == 1
        if release == "peer":
            assert sm.delete_l1_keys(keys) == (0, 1)
            lookup.free_lookup_locks(peer, 1)
            lookup.end_session(peer.request_id)
        assert sm.delete_l1_keys(keys) == (1, 0)
        lookup.end_session(key.request_id)
    finally:
        # Even a failed assertion must quiesce DMA before the L1 pool is freed.
        stream.synchronize()
        dispatch.stop()
        lookup.close()
        context.close()
