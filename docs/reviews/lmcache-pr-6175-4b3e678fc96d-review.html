"""Verify the rebuilt SDK module in a private Linux network namespace."""
import ctypes
from hashlib import sha256
import mmap
import os
from pathlib import Path
import shutil
import socket
import subprocess
import time
import uuid

import pytest

def test_default_adapter_does_not_silently_discard_isolation_config():
    from lmcache.v1.distributed.l2_adapters.mooncake_store_l2_adapter import MooncakeStoreL2AdapterConfig
    default=MooncakeStoreL2AdapterConfig.from_dict({"protocol":"tcp"})
    explicit=MooncakeStoreL2AdapterConfig.from_dict({"protocol":"tcp","process":{}})
    assert getattr(default,"process_config",None) is None
    assert getattr(explicit,"process_config",None) is not None, "explicit process configuration did not activate isolation"

def test_private_master_native_sdk_shared_arena_round_trip(tmp_path):
    from lmcache.v1.distributed.l2_adapters.mooncake_process_client import ProcessNativeClient
    from lmcache.v1.platform import create_event_notifier
    import lmcache
    native,=Path(lmcache.__file__).parent.glob("lmcache_mooncake*.so")
    assert sha256(native.read_bytes()).hexdigest()=="3d7fb1cd26f266fbc062c0f166794a5cf37e016d7794791ca015b02533ddc511"
    master=Path("/opt/glm53-image-gn/.venv/lib/python3.12/site-packages/mooncake/mooncake_master")
    log=(tmp_path/"master.log").open("w+")
    master_proc=subprocess.Popen([str(master),"--rpc_address=127.0.0.1","--rpc_port=50051","--metrics_port=50101","--enable_http_metadata_server=true","--http_metadata_server_host=127.0.0.1","--http_metadata_server_port=50808","--enable_offload=false","--logtostderr"],stdout=log,stderr=subprocess.STDOUT)
    def wait(fn, name, seconds=40):
        end=time.monotonic()+seconds
        while time.monotonic()<end:
            if master_proc.poll() is not None:
                log.flush();log.seek(0);pytest.fail("private master exited: "+log.read()[-1300:])
            value=fn()
            if value: return value
            time.sleep(.04)
        pytest.fail(name+" timed out")
    def listening():
        with socket.socket() as s:
            s.settimeout(.2)
            return s.connect_ex(("127.0.0.1",50051))==0
    notifier=create_event_notifier()
    arena=Path("/dev/shm")/("pr6175-"+uuid.uuid4().hex)
    try:
        wait(listening,"master startup")
        with arena.open("w+b") as f:
            f.truncate(1<<20)
            with mmap.mmap(f.fileno(),0,access=mmap.ACCESS_WRITE) as mapped:
                base=ctypes.addressof(ctypes.c_ubyte.from_buffer(mapped))
                address=socket.gethostbyname(socket.gethostname())
                client=ProcessNativeClient({"protocol":"tcp","local_hostname":address,"metadata_server":"http://127.0.0.1:50808/metadata","master_server_addr":"127.0.0.1:50051","global_segment_size":str(128<<20),"local_buffer_size":str(32<<20)},2,None,base,len(mapped),arena.name,notifier,startup_timeout=30,operation_timeout=15)
                try:
                    wait(lambda: client.is_ready,"native worker ready")
                    def completion(task):
                        item=wait(lambda:next((v for v in client.drain_completions() if v[0]==task),None),"SDK completion")
                        assert item[1], item
                        return item
                    payload=bytes(range(256))*16
                    mapped[4096:8192]=payload
                    with memoryview(mapped)[4096:8192] as src, memoryview(mapped)[16384:20480] as dst:
                        key="private-"+uuid.uuid4().hex
                        completion(client.submit_batch_set([key],[src]))
                        exists=completion(client.submit_batch_exists([key]))
                        assert exists[3]==[True],exists
                        completion(client.submit_batch_get([key],[dst]))
                        assert bytes(dst)==payload
                finally:
                    client.close()
                    notifier=None
    finally:
        master_proc.terminate()
        try: master_proc.wait(8)
        except subprocess.TimeoutExpired: master_proc.kill();master_proc.wait(5)
        log.close()
        if notifier is not None: notifier.close()
        arena.unlink(missing_ok=True)
