Failure mode

Another writer may fill staging memory between the initial capacity check and a valid L2 lookup result. The default controller can then miss available objects.

Compatible change

On recoverable allocation pressure, retain the plan and retry within one bounded deadline. Retain the upstream readable-L1 rule for keys that become resident while waiting. The default is zero seconds; permanent misses still fall back.

"""Compare the same released staging pressure through the public controller."""
import importlib.util
import inspect
from pathlib import Path
from threading import Event

def test_temporary_staging_can_recover_before_deadline(monkeypatch):
    spec = importlib.util.spec_from_file_location("admission_fixtures", Path.cwd() / "tests/v1/distributed/test_prefetch_controller.py")
    suite = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(suite)
    original = suite.PrefetchController
    supported = "load_admission_wait_seconds" in inspect.signature(original).parameters
    def controller(*args, **kwargs):
        if not supported:
            kwargs.pop("load_admission_wait_seconds", None)
        return original(*args, **kwargs)
    monkeypatch.setattr(suite, "PrefetchController", controller)
    failed_reservation = Event()
    reserve = suite.DramL1Manager.reserve_write
    def observed(manager, *args, **kwargs):
        result = reserve(manager,*args,**kwargs)
        if any(error.name == "OUT_OF_MEMORY" for error, _ in result.values()):
            failed_reservation.set()
        return result
    monkeypatch.setattr(suite.DramL1Manager, "reserve_write", observed)
    with suite.staging_pressure(1.0) as (l1, adapter, ctrl, request, keys, fillers):
        assert failed_reservation.wait(4), "did not observe the planned reservation under pressure"
        l1.finish_write_and_delete(fillers)
        result = suite.wait_for_result(ctrl, request)
        assert suite.row_bits(result) == [0, 1], "valid L2 cells were missed after temporary staging pressure"
        l1.finish_read(keys)
        suite.assert_l2_unlocked(adapter)
